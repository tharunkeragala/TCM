/* ============================================================================
   ENTERPRISE PLAYWRIGHT AUTOMATION PLATFORM
   Consolidated Database Schema — Single File Migration

   Target:   Microsoft SQL Server 2019+
   Database: test_case_manager
   Safe to re-run: every statement is guarded with existence checks.

   Run this whole file once (e.g. via sqlcmd, Azure Data Studio, or SSMS).
   It creates every table required by the 10 new platform features on top
   of your existing test_cases / playwright_test_runs / playwright_test_run_steps
   tables (those are assumed to already exist and are NOT redefined here).
   ============================================================================ */

USE test_case_manager;
GO

/* ============================================================================
   PHASE 1 — DATA-DRIVEN TESTING
   ============================================================================ */

-- 1.1 Data source definitions (CSV / XLSX / JSON / API) attached to a test case
IF NOT EXISTS (SELECT * FROM sys.objects WHERE object_id = OBJECT_ID(N'dbo.test_data_sources') AND type = 'U')
BEGIN
    CREATE TABLE dbo.test_data_sources (
        id                INT IDENTITY PRIMARY KEY,
        test_case_id      INT NOT NULL,
        data_source_type  VARCHAR(50)   NOT NULL,   -- CSV, XLSX, JSON, API
        source_path       NVARCHAR(MAX) NULL,       -- file path or API URL
        options           NVARCHAR(MAX) NULL,       -- JSON: {sheetName, method, headers, body}
        created_at        DATETIME      NOT NULL DEFAULT GETDATE(),
        CONSTRAINT FK_test_data_sources_test_case
            FOREIGN KEY (test_case_id) REFERENCES dbo.test_cases(id)
    );
    CREATE INDEX IX_test_data_sources_test_case ON dbo.test_data_sources(test_case_id);
END
GO

-- 1.2 Individual rows of test data loaded from a source
IF NOT EXISTS (SELECT * FROM sys.objects WHERE object_id = OBJECT_ID(N'dbo.test_data_rows') AND type = 'U')
BEGIN
    CREATE TABLE dbo.test_data_rows (
        id                 INT IDENTITY PRIMARY KEY,
        data_source_id     INT NOT NULL,
        row_number         INT NOT NULL,
        data               NVARCHAR(MAX) NOT NULL,   -- JSON blob of the row
        is_used            BIT NOT NULL DEFAULT 0,
        execution_result   VARCHAR(50) NULL,         -- PENDING, PASSED, FAILED
        created_at         DATETIME NOT NULL DEFAULT GETDATE(),
        CONSTRAINT FK_test_data_rows_source
            FOREIGN KEY (data_source_id) REFERENCES dbo.test_data_sources(id)
    );
    CREATE CLUSTERED INDEX IX_test_data_rows_source_row ON dbo.test_data_rows(data_source_id, row_number);
END
GO

-- 1.3 Mapping of {{placeholders}} in a script to a data source column
IF NOT EXISTS (SELECT * FROM sys.objects WHERE object_id = OBJECT_ID(N'dbo.test_parameter_mappings') AND type = 'U')
BEGIN
    CREATE TABLE dbo.test_parameter_mappings (
        id                     INT IDENTITY PRIMARY KEY,
        test_case_id           INT NOT NULL,
        parameter_name         VARCHAR(150) NOT NULL,   -- e.g. customer.name
        variable_placeholder   VARCHAR(150) NOT NULL,   -- e.g. {{customer.name}}
        data_column_name       VARCHAR(150) NOT NULL,   -- column in the data source
        transformation_rules   NVARCHAR(MAX) NULL,      -- JSON transformation config
        created_at             DATETIME NOT NULL DEFAULT GETDATE(),
        CONSTRAINT FK_test_parameter_mappings_test_case
            FOREIGN KEY (test_case_id) REFERENCES dbo.test_cases(id)
    );
    CREATE INDEX IX_test_parameter_mappings_test_case ON dbo.test_parameter_mappings(test_case_id);
END
GO

-- 1.4 Link between a data row, a generated run, and its iteration number
IF NOT EXISTS (SELECT * FROM sys.objects WHERE object_id = OBJECT_ID(N'dbo.test_data_execution_runs') AND type = 'U')
BEGIN
    CREATE TABLE dbo.test_data_execution_runs (
        id                 INT IDENTITY PRIMARY KEY,
        test_case_id       INT NOT NULL,
        data_row_id        INT NOT NULL,
        run_id             INT NOT NULL,
        iteration_number   INT NOT NULL,
        execution_status   VARCHAR(50) NULL,
        created_at         DATETIME NOT NULL DEFAULT GETDATE(),
        CONSTRAINT FK_test_data_execution_runs_test_case
            FOREIGN KEY (test_case_id) REFERENCES dbo.test_cases(id),
        CONSTRAINT FK_test_data_execution_runs_data_row
            FOREIGN KEY (data_row_id) REFERENCES dbo.test_data_rows(id),
        CONSTRAINT FK_test_data_execution_runs_run
            FOREIGN KEY (run_id) REFERENCES dbo.playwright_test_runs(id)
    );
    CREATE INDEX IX_test_data_execution_runs_run ON dbo.test_data_execution_runs(run_id);
END
GO

/* ============================================================================
   PHASE 1 — VARIABLE ENGINE
   ============================================================================ */

-- 1.5 Per-test-case variables (static, dynamic, environment, scoped)
IF NOT EXISTS (SELECT * FROM sys.objects WHERE object_id = OBJECT_ID(N'dbo.test_variables') AND type = 'U')
BEGIN
    CREATE TABLE dbo.test_variables (
        id                INT IDENTITY PRIMARY KEY,
        test_case_id      INT NOT NULL,
        variable_name     VARCHAR(150) NOT NULL,
        variable_type     VARCHAR(50)  NOT NULL DEFAULT 'STATIC', -- STATIC, DYNAMIC, SYSTEM, ENVIRONMENT
        variable_value    NVARCHAR(MAX) NULL,
        scope             VARCHAR(50)  NOT NULL DEFAULT 'TEST',   -- TEST, SUITE, GLOBAL
        created_at        DATETIME NOT NULL DEFAULT GETDATE(),
        CONSTRAINT FK_test_variables_test_case
            FOREIGN KEY (test_case_id) REFERENCES dbo.test_cases(id),
        CONSTRAINT UQ_test_variables_case_name UNIQUE (test_case_id, variable_name)
    );
END
GO

-- 1.6 Built-in system variable / function catalogue (uuid, timestamp, randomEmail, ...)
IF NOT EXISTS (SELECT * FROM sys.objects WHERE object_id = OBJECT_ID(N'dbo.system_variables') AND type = 'U')
BEGIN
    CREATE TABLE dbo.system_variables (
        id                   INT IDENTITY PRIMARY KEY,
        variable_name        VARCHAR(150) NOT NULL UNIQUE,
        description          NVARCHAR(MAX) NULL,
        generator_function   VARCHAR(255) NULL,   -- 'uuid', 'timestamp', 'randomEmail', etc.
        created_at           DATETIME NOT NULL DEFAULT GETDATE()
    );
END
GO

/* ============================================================================
   PHASE 2 — CONDITIONAL EXECUTION (IF / SWITCH / LOOP / WHILE / FOREACH)
   ============================================================================ */

IF NOT EXISTS (SELECT * FROM sys.objects WHERE object_id = OBJECT_ID(N'dbo.conditional_blocks') AND type = 'U')
BEGIN
    CREATE TABLE dbo.conditional_blocks (
        id                INT IDENTITY PRIMARY KEY,
        test_case_id      INT NOT NULL,
        block_type        VARCHAR(20) NOT NULL,   -- IF, SWITCH, LOOP, WHILE, FOREACH
        condition_expr    NVARCHAR(MAX) NULL,     -- expression to evaluate
        parent_block_id   INT NULL,
        sequence_order    INT NOT NULL DEFAULT 0,
        created_at        DATETIME NOT NULL DEFAULT GETDATE(),
        CONSTRAINT FK_conditional_blocks_test_case
            FOREIGN KEY (test_case_id) REFERENCES dbo.test_cases(id),
        CONSTRAINT FK_conditional_blocks_parent
            FOREIGN KEY (parent_block_id) REFERENCES dbo.conditional_blocks(id)
    );
    CREATE INDEX IX_conditional_blocks_test_case ON dbo.conditional_blocks(test_case_id, sequence_order);
END
GO

IF NOT EXISTS (SELECT * FROM sys.objects WHERE object_id = OBJECT_ID(N'dbo.conditional_block_steps') AND type = 'U')
BEGIN
    CREATE TABLE dbo.conditional_block_steps (
        id            INT IDENTITY PRIMARY KEY,
        block_id      INT NOT NULL,
        step_order    INT NOT NULL,
        step_type     VARCHAR(50) NOT NULL,   -- click, fill, api_call, keyword, ...
        step_config   NVARCHAR(MAX) NOT NULL, -- JSON payload for the step
        created_at    DATETIME NOT NULL DEFAULT GETDATE(),
        CONSTRAINT FK_conditional_block_steps_block
            FOREIGN KEY (block_id) REFERENCES dbo.conditional_blocks(id)
    );
    CREATE INDEX IX_conditional_block_steps_block ON dbo.conditional_block_steps(block_id, step_order);
END
GO

/* ============================================================================
   PHASE 2/3 — API-DRIVEN TESTING & API CHAINING
   ============================================================================ */

IF NOT EXISTS (SELECT * FROM sys.objects WHERE object_id = OBJECT_ID(N'dbo.api_endpoints') AND type = 'U')
BEGIN
    CREATE TABLE dbo.api_endpoints (
        id                       INT IDENTITY PRIMARY KEY,
        test_case_id             INT NULL,
        name                     VARCHAR(150) NOT NULL,
        method                   VARCHAR(10)  NOT NULL DEFAULT 'GET',   -- GET, POST, PUT, DELETE, PATCH
        url                      NVARCHAR(MAX) NOT NULL,
        headers                  NVARCHAR(MAX) NULL,   -- JSON
        body                     NVARCHAR(MAX) NULL,   -- JSON
        authentication_type      VARCHAR(50)  NULL,    -- NONE, BASIC, BEARER, API_KEY
        authentication_value     NVARCHAR(MAX) NULL,
        expected_status_code     INT NULL,
        assertions               NVARCHAR(MAX) NULL,   -- JSON array
        created_at               DATETIME NOT NULL DEFAULT GETDATE(),
        CONSTRAINT FK_api_endpoints_test_case
            FOREIGN KEY (test_case_id) REFERENCES dbo.test_cases(id)
    );
    CREATE INDEX IX_api_endpoints_test_case ON dbo.api_endpoints(test_case_id);
END
GO

-- Logical chain container
IF NOT EXISTS (SELECT * FROM sys.objects WHERE object_id = OBJECT_ID(N'dbo.api_chains') AND type = 'U')
BEGIN
    CREATE TABLE dbo.api_chains (
        id              INT IDENTITY PRIMARY KEY,
        test_case_id    INT NOT NULL,
        name            VARCHAR(150) NOT NULL,
        description     NVARCHAR(MAX) NULL,
        created_at      DATETIME NOT NULL DEFAULT GETDATE(),
        CONSTRAINT FK_api_chains_test_case
            FOREIGN KEY (test_case_id) REFERENCES dbo.test_cases(id)
    );
END
GO

-- Individual ordered steps of a chain, each pointing at one api_endpoint
IF NOT EXISTS (SELECT * FROM sys.objects WHERE object_id = OBJECT_ID(N'dbo.api_chain_steps') AND type = 'U')
BEGIN
    CREATE TABLE dbo.api_chain_steps (
        id                INT IDENTITY PRIMARY KEY,
        api_chain_id      INT NOT NULL,
        api_endpoint_id   INT NOT NULL,
        step_order        INT NOT NULL,
        extract_rules     NVARCHAR(MAX) NULL,   -- JSON: {"token": "$.access_token"}
        pass_to_next      NVARCHAR(MAX) NULL,   -- JSON: which vars to forward
        retry_policy      NVARCHAR(MAX) NULL,   -- JSON: {maxRetries, delayMs}
        stop_on_failure   BIT NOT NULL DEFAULT 1,
        created_at        DATETIME NOT NULL DEFAULT GETDATE(),
        CONSTRAINT FK_api_chain_steps_chain
            FOREIGN KEY (api_chain_id) REFERENCES dbo.api_chains(id),
        CONSTRAINT FK_api_chain_steps_endpoint
            FOREIGN KEY (api_endpoint_id) REFERENCES dbo.api_endpoints(id)
    );
    CREATE INDEX IX_api_chain_steps_chain ON dbo.api_chain_steps(api_chain_id, step_order);
END
GO

-- Captured request/response per run, for reporting and debugging
IF NOT EXISTS (SELECT * FROM sys.objects WHERE object_id = OBJECT_ID(N'dbo.api_responses') AND type = 'U')
BEGIN
    CREATE TABLE dbo.api_responses (
        id                INT IDENTITY PRIMARY KEY,
        run_id            INT NOT NULL,
        api_endpoint_id   INT NOT NULL,
        request_body      NVARCHAR(MAX) NULL,
        response_body     NVARCHAR(MAX) NULL,
        status_code       INT NULL,
        headers           NVARCHAR(MAX) NULL,
        duration_ms       INT NULL,
        created_at        DATETIME NOT NULL DEFAULT GETDATE(),
        CONSTRAINT FK_api_responses_run
            FOREIGN KEY (run_id) REFERENCES dbo.playwright_test_runs(id),
        CONSTRAINT FK_api_responses_endpoint
            FOREIGN KEY (api_endpoint_id) REFERENCES dbo.api_endpoints(id)
    );
    CREATE INDEX IX_api_responses_run ON dbo.api_responses(run_id);
END
GO

/* ============================================================================
   PHASE 4 — SELF-HEALING LOCATORS
   ============================================================================ */

IF NOT EXISTS (SELECT * FROM sys.objects WHERE object_id = OBJECT_ID(N'dbo.locator_history') AND type = 'U')
BEGIN
    CREATE TABLE dbo.locator_history (
        id                     INT IDENTITY PRIMARY KEY,
        test_case_id           INT NOT NULL,
        original_locator       NVARCHAR(MAX) NOT NULL,
        actual_locator_used    NVARCHAR(MAX) NULL,       -- selector that actually worked (may equal original)
        alternative_locators   NVARCHAR(MAX) NULL,       -- JSON array of candidates generated
        was_healed             BIT NOT NULL DEFAULT 0,
        healing_reason         VARCHAR(50) NULL,         -- ELEMENT_MOVED, ID_CHANGED, SELF_HEALED, MANUAL_UPDATE
        stability_score        FLOAT NULL,
        use_count              INT NOT NULL DEFAULT 1,
        last_used              DATETIME NOT NULL DEFAULT GETDATE(),
        created_at             DATETIME NOT NULL DEFAULT GETDATE(),
        CONSTRAINT FK_locator_history_test_case
            FOREIGN KEY (test_case_id) REFERENCES dbo.test_cases(id)
    );
    CREATE INDEX IX_locator_history_test_case ON dbo.locator_history(test_case_id, stability_score);
END
GO
