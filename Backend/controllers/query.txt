-- ============================================================
-- Flow Diagrams feature
-- One diagram (draft) per project, with an explicit, immutable
-- version history table. Editing autosaves into project_diagrams
-- silently; "Save as New Version" snapshots that state into
-- project_diagram_versions.
-- ============================================================

USE test_case_manager;
GO

-- Draft / current working state — one row per project.
IF NOT EXISTS (
  SELECT 1 FROM sys.tables WHERE name = 'project_diagrams'
)
BEGIN
  CREATE TABLE dbo.project_diagrams (
    id                     INT IDENTITY(1,1) PRIMARY KEY,
    project_id             INT NOT NULL,
    nodes                  NVARCHAR(MAX) NOT NULL DEFAULT '[]',
    edges                  NVARCHAR(MAX) NOT NULL DEFAULT '[]',
    viewport               NVARCHAR(MAX) NULL,
    latest_version_number  INT NOT NULL DEFAULT 0,
    created_by             INT NULL,
    created_at             DATETIME NOT NULL DEFAULT GETDATE(),
    updated_by             INT NULL,
    updated_at             DATETIME NOT NULL DEFAULT GETDATE(),
    CONSTRAINT UQ_project_diagrams_project UNIQUE (project_id),
    CONSTRAINT FK_project_diagrams_project FOREIGN KEY (project_id)
      REFERENCES dbo.projects(id)
  );
END
GO

-- Immutable version snapshots.
IF NOT EXISTS (
  SELECT 1 FROM sys.tables WHERE name = 'project_diagram_versions'
)
BEGIN
  CREATE TABLE dbo.project_diagram_versions (
    id              INT IDENTITY(1,1) PRIMARY KEY,
    project_id      INT NOT NULL,
    version_number  INT NOT NULL,
    label           VARCHAR(150) NULL,
    nodes           NVARCHAR(MAX) NOT NULL,
    edges           NVARCHAR(MAX) NOT NULL,
    viewport        NVARCHAR(MAX) NULL,
    created_by      INT NULL,
    created_at      DATETIME NOT NULL DEFAULT GETDATE(),
    CONSTRAINT UQ_project_diagram_versions UNIQUE (project_id, version_number),
    CONSTRAINT FK_project_diagram_versions_project FOREIGN KEY (project_id)
      REFERENCES dbo.projects(id)
  );

  CREATE INDEX IX_project_diagram_versions_project
    ON dbo.project_diagram_versions (project_id, version_number DESC);
END
GO
